fit f(x) 'powerlawTL_IC_s1_K_3.2e+04.dat' using 2:(log10($4)) via b,a
a2 = a

fit f(x) 'powerlawTL_IC_s1_K_1.0e+05.dat' using 2:(log10($4)) via b,a
a3 = a

powerlaw = (a1 + a2 + a3 + a4 + a5)/5.0
stdev = sqrt(( \
    (a1-powerlaw)**2 + (a2-powerlaw)**2 + \
    (a3-powerlaw)**2 + (a4-powerlaw)**2 + \
    (a5-powerlaw)**2 \
    ) / 4.0)
